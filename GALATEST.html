/* ---------------------- CALCULAR PRÓXIMO MARTES ---------------------- */ /** * getNextTuesday() * Calcula el próximo martes a partir de hoy. * Si hoy es martes, devuelve hoy. * Si hoy es miércoles o después, devuelve el martes de la próxima semana. */ function getNextTuesday() { const today = new Date(); today.setHours(0, 0, 0, 0); const dayOfWeek = today.getDay(); const daysUntilTuesday = (2 - dayOfWeek + 7) % 7; const daysToAdd = (daysUntilTuesday === 0) ? 0 : daysUntilTuesday; const nextTuesday = new Date(today); nextTuesday.setDate(nextTuesday.getDate() + daysToAdd); return nextTuesday; } /* ---------------------- EXTRAER Y FILTRAR HABITACIONES ---------------------- */ /** * extractRoomsFromExternalSpreadsheet(url) * * ESTRUCTURA DEL ARCHIVO FILTRADO: * A: Hotel * B: Habitación (números como 101, 202, etc.) * C: Apellidos * D: Nombre * E: Salida (fecha en serie como 46176) * F: (vacío) * G: (vacío) * H: Entrada * * LÓGICA: * 1. Lee el próximo martes desde hoy * 2. Por cada huésped (fila 2 en adelante): * - Lee habitación (B, índice 1) * - Lee fecha de salida/checkout (E, índice 4) * - SI checkout > próximo martes → INCLUIR * - SI checkout <= próximo martes → EXCLUIR * 3. Elimina duplicados * 4. Devuelve array de habitaciones únicas */ function extractRoomsFromExternalSpreadsheet(url) { try { if (!url || typeof url !== 'string') { Logger.log('extractRoomsFromExternalSpreadsheet: URL inválida'); return []; } // Abrir el archivo compartido const file = SpreadsheetApp.openByUrl(url); const sheet = file.getActiveSheet(); // Obtener todos los datos const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); Logger.log('=== DEBUG: Hoja abierta ==='); Logger.log('Última fila: ' + lastRow); Logger.log('Última columna: ' + lastCol); if (lastRow < 2 || lastCol < 5) { Logger.log('extractRoomsFromExternalSpreadsheet: hoja muy pequeña o vacía (necesita mínimo fila 2 y columna E)'); return []; } // Obtener el próximo martes desde hoy const nextTuesday = getNextTuesday(); Logger.log('Próximo martes: ' + formatDateDDMMYYYY(nextTuesday)); // Leer datos de la hoja (desde fila 2) const data = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues(); const rooms = []; const seen = {}; let processedCount = 0; let includedCount = 0; let excludedCount = 0; for (let row = 0; row < data.length; row++) { try { const habRow = data[row]; processedCount++; // Columna B (índice 1): número de habitación const roomNum = (habRow[1] + '').trim(); if (!roomNum) { Logger.log('Fila ' + (row + 2) + ': Sin habitación (B vacío) → EXCLUIDA'); excludedCount++; continue; // Sin habitación, omitir } // Columna E (índice 4): fecha de salida/checkout let checkoutDate = habRow[4]; if (!checkoutDate) { Logger.log('Fila ' + (row + 2) + ': Hab ' + roomNum + ' → Sin checkout (E vacío) → EXCLUIDA'); excludedCount++; continue; // Sin fecha de checkout, omitir } // Parsear fecha de checkout const parsed = parsePossibleDate(checkoutDate); if (!parsed) { Logger.log('Fila ' + (row + 2) + ': Hab ' + roomNum + ' → Fecha inválida: ' + checkoutDate + ' → EXCLUIDA'); excludedCount++; continue; // Fecha inválida, omitir } // LÓGICA DE FILTRADO POR MARTES: // Incluir si checkout > próximo martes (sale DESPUÉS del martes) // Excluir si checkout <= próximo martes (sale el martes o antes) if (parsed <= nextTuesday) { Logger.log('Fila ' + (row + 2) + ': Hab ' + roomNum + ' → Checkout ' + formatDateDDMMYYYY(parsed) + ' <= ' + formatDateDDMMYYYY(nextTuesday) + ' → EXCLUIDA'); excludedCount++; continue; // Excluir } // No es duplicado → agregar if (!seen[roomNum]) { rooms.push(roomNum); seen[roomNum] = true; includedCount++; Logger.log('Fila ' + (row + 2) + ': Hab ' + roomNum + ' → Checkout ' + formatDateDDMMYYYY(parsed) + ' > ' + formatDateDDMMYYYY(nextTuesday) + ' → ✓ INCLUIDA'); } else { Logger.log('Fila ' + (row + 2) + ': Hab ' + roomNum + ' → DUPLICADO (ya existe) → EXCLUIDA'); excludedCount++; } } catch (rowErr) { Logger.log('extractRoomsFromExternalSpreadsheet: error procesando fila ' + (row + 2) + ': ' + rowErr); excludedCount++; continue; } } Logger.log('=== RESUMEN ==='); Logger.log('Filas procesadas: ' + processedCount); Logger.log('Habitaciones incluidas: ' + includedCount); Logger.log('Filas excluidas: ' + excludedCount); Logger.log('Habitaciones únicas totales: ' + rooms.length); if (rooms.length > 0 && rooms.length <= 50) { Logger.log('Habitaciones: ' + rooms.join(', ')); } return rooms; } catch (err) { Logger.log('extractRoomsFromExternalSpreadsheet ERROR GENERAL: ' + err); return []; } }
